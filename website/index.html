}

from google.cloud import storage, pubsub_v1

# Connect to Aura-GCP storage
client = storage.Client(client_options={"api_endpoint": "http://localhost:8080"})
bucket = client.create_bucket("mi-data-lake")
print("Bucket created locally:", bucket.name)

# Connect to Aura-GCP Pub/Sub
pub = pubsub_v1.PublisherClient(client_options={"api_endpoint": "localhost:8080"})
topic_path = pub.topic_path("local-project", "events")
pub.create_topic(request={"name": topic_path})
import { Storage } from "@google-cloud/storage";

const storage = new Storage({
  apiEndpoint: "http://localhost:8080",
  projectId: "local-project",
});

const [bucket] = await storage.createBucket("node-local-bucket");
console.log("GCS Bucket created in < 1ms:", bucket.name);
package main

import (
    "context"
    "fmt"
    "cloud.google.com/go/storage"
    "google.golang.org/api/option"
)

func main() {
    ctx := context.Background()
    client, _ := storage.NewClient(ctx,
        option.WithEndpoint("http://localhost:8080/storage/v1/"),
        option.WithoutAuthentication(),
    )
    bkt := client.Bucket("go-local-bucket")
    _ = bkt.Create(ctx, "local-project", nil)
    fmt.Println("Bucket created successfully on Aura-GCP!")
}
